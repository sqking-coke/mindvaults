# 05 - 数据库设计

> 状态：✅ 已实施 | 版本：v2.1 多知识库架构 | 迁移：0004 → 0008
>
> 本文档涵盖数据库 DDL 设计、索引策略、检索查询、Alembic 迁移、ORM 模型、Schema、Service、API 路由、前端对接及测试规格，是数据库层的**唯一真相来源**。

---

## 1. 设计原则

- **KB 级隔离**：文档 → 切片 → 会话 → QA 记录，通过 `kb_id` 层层归属
- **每 KB 独立配置**：`kb_config` 以 `kb_id` 作 PK+FK，一对一
- **级联删除**：删 KB → 级联删文档 → 级联删切片
- **无软删除扩散**：只有 `kb_documents.deleted_at` 用软删除，其他物理删除
- **预留扩展**：`kb_data_sources` / `kb_raw_entries` / `kb_feedback` 三张表本期建好，后期接定时任务和反馈学习

---

## 2. ER 图

```
kb_knowledge_bases (1) ──── (1) kb_config          ← KB + 独立配置
       │
       │ (1:N)
       ├── kb_documents (N) ──── (1:N) kb_chunks ← 文档 + 切片
       │
       │ (1:N)
       ├── kb_sessions (N) ──── (1:N) kb_qa_records  ← 会话 + QA
       │
       │ (1:N)
       ├── kb_data_sources (N) ─── (1:N) kb_raw_entries ← 外部数据源（预留）
       │
       └── kb_feedback (N)                          ← 反馈学习（预留）
```

```sql

    kb_id               BIGINT PRIMARY KEY REFERENCES kb_knowledge_bases(id) ON DELETE CASCADE,
    embedding_dim       INTEGER NOT NULL DEFAULT 1024,
    chunk_size          INTEGER NOT NULL DEFAULT 500,
    chunk_overlap       INTEGER NOT NULL DEFAULT 50,
    top_k               INTEGER NOT NULL DEFAULT 5,
    similarity_threshold FLOAT NOT NULL DEFAULT 0.35,

    -- LLM 配置
    llm_provider        VARCHAR(50),        -- ollama / openai
    llm_base_url        VARCHAR(255),
    llm_model           VARCHAR(100),
    llm_api_key         VARCHAR(255),
```

### 3.3 文档表

```sql
CREATE TABLE kb_documents (
    id          BIGSERIAL PRIMARY KEY,
    kb_id       BIGINT NOT NULL REFERENCES kb_knowledge_bases(id) ON DELETE CASCADE,
    doc_name    VARCHAR(255) NOT NULL,
    doc_type    VARCHAR(10) NOT NULL,        -- txt / md / pdf / docx / doc
    doc_desc    TEXT,
    file_path   VARCHAR(500) NOT NULL,
    status      SMALLINT NOT NULL DEFAULT 1, -- 0=失败 1=解析中 2=成功 3=禁用
    chunk_count INTEGER NOT NULL DEFAULT 0,
    source      VARCHAR(50),                 -- obsidian / upload / api (数据来源)
    file_size   BIGINT NOT NULL DEFAULT 0,   -- 物理文件字节数，上传时 stat 写入
    status_detail JSONB NOT NULL DEFAULT '{}', -- 摄入心跳: {"phase":"embedding","total":200}
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    deleted_at  TIMESTAMPTZ                  -- 软删除
);

CREATE INDEX idx_documents_kb_id ON kb_documents(kb_id);
CREATE INDEX idx_documents_status ON kb_documents(status);
CREATE INDEX idx_documents_deleted ON kb_documents(deleted_at);
```

## 4. 索引策略

| 表 | 索引 | 类型 | 用途 |
|----|------|------|------|
| kb_documents | `kb_id` | B-tree | 按 KB 过滤文档 |
| kb_documents | `status` | B-tree | 按状态过滤 |
| kb_documents | `deleted_at` | B-tree | 排除已删除 |
| kb_chunks | `document_id` | B-tree | 按文档查切片 |
| kb_chunks | `embedding` | **HNSW** | 向量相似度检索 |
| kb_sessions | `session_id` | UNIQUE B-tree | UUID 查会话 |
| kb_sessions | `kb_id` | B-tree | 按 KB 隔离会话 |
| kb_qa_records | `session_id` | B-tree | 按会话查历史 |
| kb_qa_records | `created_at` | B-tree | 时序查询/统计 |

## 5. 检索查询（核心 SQL）

```sql
-- 限定 KB 的向量检索
SET hnsw.ef_search = 40;

SELECT
    c.id AS chunk_id,
    d.doc_name,
    c.content,
    1 - (c.embedding <=> :query_vec) AS similarity,
    c.page
FROM kb_chunks c
JOIN kb_documents d ON c.document_id = d.id
WHERE d.deleted_at IS NULL
  AND d.status = 2                    -- COMPLETED
  AND d.kb_id = :kb_id               -- ★ KB 隔离
  AND 1 - (c.embedding <=> :query_vec) >= :threshold
ORDER BY similarity DESC
LIMIT :top_k;
```

```python
"""知识库 CRUD 服务层。"""
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.core.exceptions import NotFoundError
from app.models.knowledge_base import KnowledgeBase
from app.models.config import KbConfig
from app.models.document import KbDocument
from app.schemas.knowledge_base import (
    KbCreateRequest, KbUpdateRequest, KbConfigRequest,
)


async def create_kb(db: AsyncSession, req: KbCreateRequest) -> KnowledgeBase:
    """创建知识库，同时初始化一行默认 kb_config。"""
    kb = KnowledgeBase(name=req.name, description=req.description or "")
    db.add(kb)
    await db.flush()
    config = KbConfig(kb_id=kb.id)  # 全部使用列默认值
    db.add(config)
    await db.commit()
    await db.refresh(kb)
    return kb


async def list_kbs(db: AsyncSession) -> list[dict]:
    """列出所有知识库，附带文档计数。"""
    rows = (await db.execute(
        select(
            KnowledgeBase,
            func.count(KbDocument.id).label("doc_count"),
        )
        .outerjoin(KbDocument, KbDocument.kb_id == KnowledgeBase.id)
        .where(KbDocument.deleted_at.is_(None))
        .group_by(KnowledgeBase.id)
        .order_by(KnowledgeBase.id)
    )).all()

    return [
        {
            "id": kb.id, "name": kb.name,
            "description": kb.description or "",
            "doc_count": doc_count,
            "created_at": kb.created_at, "updated_at": kb.updated_at,
        }
        for kb, doc_count in rows
    ]


async def get_kb(db: AsyncSession, kb_id: int) -> KnowledgeBase:
    kb = await db.get(KnowledgeBase, kb_id)
    if kb is None:
        raise NotFoundError(f"知识库不存在: {kb_id}")
    return kb


async def update_kb(db: AsyncSession, kb_id: int, req: KbUpdateRequest) -> KnowledgeBase:
    kb = await get_kb(db, kb_id)
    if req.name is not None:
        kb.name = req.name
    if req.description is not None:
        kb.description = req.description
    await db.commit()
    await db.refresh(kb)
    return kb


async def delete_kb(db: AsyncSession, kb_id: int) -> None:
    """级联删除：KB → 文档 → 切片 + 会话 → QA 记录（由 DB FK CASCADE 保证）。"""
    kb = await get_kb(db, kb_id)
    await db.delete(kb)
    await db.commit()


async def get_kb_config(db: AsyncSession, kb_id: int) -> KbConfig:
    """获取指定 KB 的配置，不存在则自动创建默认配置。"""
    row = (await db.execute(
        select(KbConfig).where(KbConfig.kb_id == kb_id)
    )).scalar_one_or_none()
    if row is None:
        row = KbConfig(kb_id=kb_id)
        db.add(row)
        await db.flush()
        await db.commit()
    return row


async def update_kb_config(
    db: AsyncSession, kb_id: int, req: KbConfigRequest
) -> KbConfig:
    cfg = await get_kb_config(db, kb_id)
    if req.chunk_size is not None:
        cfg.chunk_size = req.chunk_size
    if req.chunk_overlap is not None:
        cfg.chunk_overlap = req.chunk_overlap
    if req.top_k is not None:
        cfg.top_k = req.top_k
    if req.similarity_threshold is not None:
        cfg.similarity_threshold = req.similarity_threshold
    if req.llm_provider is not None:
        cfg.llm_provider = req.llm_provider.strip().lower()
    if req.llm_base_url is not None:
        cfg.llm_base_url = req.llm_base_url.strip()
    if req.llm_model is not None:
        cfg.llm_model = req.llm_model.strip()
    if req.llm_api_key is not None:
        key = req.llm_api_key.strip()
        if key and "••" not in key:
            cfg.llm_api_key = key
        elif not key:
            cfg.llm_api_key = ""
    if req.llm_temperature is not None:
        cfg.llm_temperature = req.llm_temperature
    if req.system_prompt is not None:
        cfg.system_prompt = req.system_prompt.strip()
    await db.commit()
    await db.refresh(cfg)
    return cfg
```

```python
# _pgvector_search() 加 kb_id 参数
async def _pgvector_search(
    db: AsyncSession,
    query_embedding: list[float],
    k: int,
    thresh: float,
    kb_id: int,  # ★ NEW
) -> list[RefChunk]:
    vec = type_coerce(query_embedding, Vector(1024))
    stmt = (
        select(KbChunk.id.label("chunk_id"), KbDocument.doc_name,
               KbChunk.content,
               (1.0 - func.cosine_distance(KbChunk.embedding, vec)).label("similarity"),
               KbChunk.page)
        .join(KbDocument, KbChunk.document_id == KbDocument.id)
        .where(
            KbDocument.deleted_at.is_(None),
            KbDocument.status == 2,
            KbDocument.kb_id == kb_id,  # ★ KB 隔离
            func.cosine_distance(KbChunk.embedding, vec) <= 1.0 - thresh,
        )
        .order_by(similarity_expr.desc())
        .limit(k)
    )
    rows = (await db.execute(stmt)).all()
    return [
        RefChunk(chunk_id=r.chunk_id, doc_name=r.doc_name,
                 content=r.content, similarity=round(r.similarity, 4), page=r.page)
        for r in rows
    ]

# retrieve_chunks() 签名变更
async def retrieve_chunks(
    db: AsyncSession,
    query_embedding: list[float],
    kb_id: int | None = None,  # ★ NEW
    top_k: int | None = None,
    threshold: float | None = None,
) -> list[RefChunk]:
    cfg = await get_config_by_kb(db, kb_id or 1)
    k = top_k if top_k is not None else cfg.top_k
    thresh = threshold if threshold is not None else cfg.similarity_threshold
    return await _pgvector_search(db, query_embedding, k, thresh, kb_id or 1)
```

## 15. 风险与回滚

| 风险 | 等级 | 应对 |
|------|------|------|
| kb_config 主键变更失败 | 中 | 迁移前 `pg_dump kb_config` 备份；失败后手动恢复 |
| 现有功能回归 | 中 | 保留 `get_config()` 旧函数作向后兼容；`kb_id=None` 时默认 KB 1 |
| 前端 mock KB 数据丢失 | 低 | 迁移前检查前端 KB 列表，手动在后端创建同名 KB |
| 性能退化（检索加 JOIN） | 低 | `kb_documents.kb_id` 已有索引，WHERE 条件走 Index Scan |

## 17. 应用层事务规范

> 补充于 2026-05-31

### 核心原则

**任何 `db.add()` + `db.flush()` 后，所有代码路径必须显式 `commit()` 或 `rollback()`，不得依赖会话垃圾回收。**

SQLAlchemy async session 的 `close()` 会回滚未提交事务，但该时机不可控（依赖 ASGI 任务回收），期间持有的行级锁会阻塞后续请求。

### 典型 Bug 模式与修复

**模式 1：early return 未提交**

```python
# ❌ db.flush() 后直接 return → 行级锁未释放
db.add(record)
await db.flush()
if error:
    return  # 💀

# ✅ 每个 return 前显式提交或回滚
db.add(record)
await db.flush()
if error:
    await db.commit()  # 或 rollback()
    return
```

**模式 2：生成器中长时间持锁**

```python
# ❌ flush 后等 30s+ 才 commit
db.add(session)
await db.flush()
result = await llm_call()   # 一直持锁
await db.commit()            # 太晚了

# ✅ 外部调用前立即提交
db.add(session)
await db.flush()
await db.commit()            # 释放锁
result = await llm_call()    # 新事务，不持锁
```

**模式 3：循环中 flush 失败后未回滚**

```python
# ❌ flush 失败后会话损坏
for item in items:
    try:
        db.add(record)
        await db.flush()
    except Exception:
        failed += 1           # 💀 未回滚

# ✅ 异常时回滚重置会话
for item in items:
    try:
        db.add(record)
        await db.flush()
    except Exception:
        await db.rollback()
        failed += 1
```

**模式 4：getter 中的隐式写入未提交**

```python
# ❌ 创建默认行后未提交 → GET 端点调用者不提交
async def get_config(db, kb_id):
    row = await db.execute(select(...))
    if row is None:
        row = KbConfig(kb_id=kb_id)
        db.add(row)
        await db.flush()      # 💀

# ✅ 自包含事务
async def get_config(db, kb_id):
    row = await db.execute(select(...))
    if row is None:
        row = KbConfig(kb_id=kb_id)
        db.add(row)
        await db.flush()
        await db.commit()
    return row
```
