### 5.1 新增：`app/services/kb_service.py` — 完整实现

```python
"""知识库 CRUD 服务层。"""
from sqlalchemy import select, func, delete
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.core.exceptions import NotFoundError
from app.models.knowledge_base import KnowledgeBase
from app.models.config import KbConfig
from app.models.document import KbDocument
from app.models.session import KbSession
from app.models.qa_record import KbQaRecord
from app.models.chunk import KbChunk
from app.schemas.knowledge_base import (
    KbCreateRequest, KbUpdateRequest, KbInfo,
    KbConfigRequest, KbConfigResponse,
)


async def create_kb(db: AsyncSession, req: KbCreateRequest) -> KnowledgeBase:
    """创建知识库，同时初始化一行默认 kb_config。"""
    kb = KnowledgeBase(name=req.name, description=req.description or "")
    db.add(kb)
    await db.flush()
    config = KbConfig(kb_id=kb.id)  # 全部使用列默认值
    db.add(config)
    await db.commit()
    await db.refresh(kb)
    return kb


async def list_kbs(db: AsyncSession) -> list[dict]:
    """列出所有知识库，附带文档计数。"""
    rows = (await db.execute(
        select(
            KnowledgeBase,
            func.count(KbDocument.id).label("doc_count"),
        )
        .outerjoin(KbDocument, KbDocument.kb_id == KnowledgeBase.id)
        .where(KbDocument.deleted_at.is_(None))
        .group_by(KnowledgeBase.id)
        .order_by(KnowledgeBase.id)
    )).all()

    return [
        {
            "id": kb.id,
            "name": kb.name,
            "description": kb.description or "",
            "doc_count": doc_count,
            "created_at": kb.created_at,
            "updated_at": kb.updated_at,
        }
        for kb, doc_count in rows
    ]


async def get_kb(db: AsyncSession, kb_id: int) -> KnowledgeBase:
    kb = await db.get(KnowledgeBase, kb_id)
    if kb is None:
        raise NotFoundError(f"知识库不存在: {kb_id}")
    return kb


async def update_kb(db: AsyncSession, kb_id: int, req: KbUpdateRequest) -> KnowledgeBase:
    kb = await get_kb(db, kb_id)
    if req.name is not None:
        kb.name = req.name
    if req.description is not None:
        kb.description = req.description
    await db.commit()
    await db.refresh(kb)
    return kb


async def delete_kb(db: AsyncSession, kb_id: int) -> None:
    """级联删除：KB → 文档 → 切片 + 会话 → QA 记录。
    SQLAlchemy 模型上的 ondelete="CASCADE" 确保了数据库级自动级联。
    """
    kb = await get_kb(db, kb_id)
    # 手动清理不在 CASCADE 链上的关联（chunk 通过 document FK CASCADE）
    # 但 kb_chunks 的 FK 是 document_id，document 有 kb_id FK CASCADE
    # 所以删 kb → 自动删 document → 自动删 chunk ✓
    # sessions → qa_records 同理 ✓
    await db.delete(kb)
    await db.commit()


async def get_kb_config(db: AsyncSession, kb_id: int) -> KbConfig:
    """获取指定 KB 的配置，不存在则自动创建默认配置。"""
    row = (await db.execute(
        select(KbConfig).where(KbConfig.kb_id == kb_id)
    )).scalar_one_or_none()
    if row is None:
        row = KbConfig(kb_id=kb_id)
        db.add(row)
        await db.flush()
    return row


async def update_kb_config(
    db: AsyncSession, kb_id: int, req: KbConfigRequest
) -> KbConfig:
    cfg = await get_kb_config(db, kb_id)

    # RAG 参数
    if req.chunk_size is not None:
        cfg.chunk_size = req.chunk_size
    if req.chunk_overlap is not None:
        cfg.chunk_overlap = req.chunk_overlap
```

#### `_pgvector_search()` 加 `kb_id` 参数

```python
async def _pgvector_search(
    db: AsyncSession,
    query_embedding: list[float],
    k: int,
    thresh: float,
    kb_id: int,  # ★ NEW
) -> list[RefChunk]:
    vec = type_coerce(query_embedding, Vector(1024))
    similarity_expr = 1.0 - func.cosine_distance(KbChunk.embedding, vec)

    stmt = (
        select(
            KbChunk.id.label("chunk_id"),
            KbDocument.doc_name,
            KbChunk.content,
            similarity_expr.label("similarity"),
            KbChunk.page,
        )
        .join(KbDocument, KbChunk.document_id == KbDocument.id)
        .where(
            KbDocument.deleted_at.is_(None),
            KbDocument.status == 2,           # DOC_STATUS_COMPLETED
            KbDocument.kb_id == kb_id,        # ★ KB 隔离
            func.cosine_distance(KbChunk.embedding, vec) <= 1.0 - thresh,
        )
        .order_by(similarity_expr.desc())
        .limit(k)
    )

    rows = (await db.execute(stmt)).all()
    return [
        RefChunk(chunk_id=r.chunk_id, doc_name=r.doc_name,
                 content=r.content, similarity=round(r.similarity, 4), page=r.page)
        for r in rows
    ]
```

## 10. 风险与回滚

| 风险 | 等级 | 应对 |
|------|------|------|
| kb_config 主键变更失败 | 中 | 迁移前 `pg_dump kb_config` 备份；失败后手动恢复 |
| 现有功能回归 | 中 | 保留 `get_config()` 旧函数作向后兼容；`kb_id=None` 时默认 KB 1 |
| 前端 mock KB 数据丢失 | 低 | 迁移前检查前端 KB 列表，手动在后端创建同名 KB |
| 性能退化（检索加 JOIN） | 低 | `kb_documents.kb_id` 已有索引，WHERE 条件走 Index Scan |
